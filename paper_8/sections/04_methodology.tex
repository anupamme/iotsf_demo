% ============================================================
\section{Fine-Tuning Protocol}
\label{sec:method}
% ============================================================

\textbf{Moirai for forecasting and anomaly scoring.}
Moirai~\cite{woo2024moirai} predicts a probabilistic distribution
over a target horizon given context.  For ETT forecasting we report
MSE on the median of 20 forecast samples at $h{=}96$ and $h{=}192$
(Appendix~\ref{app:moirai_median}).  For IoT the anomaly score is
the fraction of timesteps outside the $1{-}\alpha$ CI
($\alpha{=}0.05$) over a 32-timestep horizon.

\textbf{Fine-tuning objective.}
Condition~B uses NLL only.  Condition~C adds supervised
contrastive loss~\cite{khosla2020supcon}:
$\mathcal{L}(\theta) = \mathcal{L}_{\text{NLL}}(\theta) +
\lambda\,\mathcal{L}_{\text{SupCon}}(\mathbf{z}, \mathbf{y};\, \tau)$,
with $\lambda = 0.5$ and $\tau = 0.07$.
For ETT, temporal clusters (8 bins by time-series position) serve
as SupCon label.

\textbf{Training details.}
AdamW ($\eta = 10^{-4}$, weight decay $10^{-2}$), batch size 32,
early stopping with patience 3, gradient clipping at 1.0.  ETT
fine-tuning runs 20 epochs at $n{=}500$ (10 seeds) and is swept
across $n{\in}\{500,1{\text{k}},2{\text{k}},5{\text{k}},10{\text{k}}\}$.
All Moirai sizes (Small 13.8M, Base 91M, Large 311M) use the same
protocol.

\textbf{Deterministic training protocol.}
All $n{=}10$k per-seed results use CUDA-deterministic training:
\texttt{CUBLAS\_WORKSPACE\_CONFIG=:4096:8}, \texttt{torch.backends.cudnn.deterministic=True},
seeded DataLoader workers, and fixed \texttt{PYTHONHASHSEED}.
This makes the 10-seed per-seed comparison reproducible: any researcher
can recover identical per-seed forgetting values by
re-running with the same seeds on the same GPU architecture (A10G).
Full seed list, per-seed values, and hardware specification are in
Appendix~\ref{app:probing}.
Cross-hardware reproducibility (e.g., A100 vs.\ A10G) has not been empirically
verified; the deterministic guarantee applies within the same GPU architecture
(A10G, as used in this paper).

\textbf{uni2ts PackedStdScaler patch.\footnote{We patched an in-place
tensor mutation in \texttt{uni2ts}'s \texttt{PackedStdScaler} that
silently detaches autograd; a differentiable \texttt{torch.where}
replaces the in-place write. All results use the patched scaler
(released with our code). See Appendix~\ref{app:scaler_patch} for details.}}
\textbf{Any Moirai fine-tuning reproduction of this paper requires the patch.}

\textbf{Representation diagnostics.}
We track representation change with three metrics:
(1)~CKA~\cite{kornblith2019cka} between pre-trained and fine-tuned
encoder representations (geometric similarity);
and (2)~weight drift ($\ell_2$ distance from pre-trained parameters).
An earlier version added linear-probe $\Delta R^2$ as a third diagnostic;
because its absolute $R^2(\text{FT})$ is deeply negative, it measures
relative change in a regime where the probe does not generalise, and we
retract it (\S\ref{sec:retraction}).

\begin{table}[h]
\centering\small
\caption{Notation summary for probe-related quantities.}
\label{tab:notation}
\begin{tabular}{ll}
\toprule
Symbol & Definition \\
\midrule
$R^2_{\text{task}}(\text{PT})$ & $1 - \text{MSE}_{\text{ZS}}/\text{MSE}_{\text{Linear}}$; positive $\Rightarrow$ gate-pass \\
$R^2(\text{PT})$, $R^2(\text{FT})$ & Ridge probe score on pre-trained / fine-tuned encoder (retracted axis) \\
$\Delta R^2$ & $R^2(\text{FT}) - R^2(\text{PT})$; \textbf{retracted}, see \S\ref{sec:retraction} \\

\bottomrule
\end{tabular}
\end{table}

\textbf{Rationale for experimental conditions.}
Each condition (used throughout \S\ref{sec:forecasting}--\S\ref{sec:analysis})
answers one question, spanning the axis from unconstrained fine-tuning
(maximum drift) to a frozen encoder (zero drift); Table~\ref{tab:conditions}
makes the mapping explicit.  This lets us attribute any observed effect to a
specific intervention rather than to fine-tuning in the aggregate.

\caption{Experimental conditions and the question each isolates.}
\label{tab:conditions}
\begin{tabular}{lll}
\toprule
Cond. & Setting & Question it isolates \\
\midrule
B & NLL-only fine-tune & Baseline: how much does unconstrained fine-tuning drift/forget? \\
C & NLL $+$ SupCon & Does an explicit representation-shaping loss prevent drift? \\
D & Frozen encoder & Control: separates encoder updates from head training / overfitting \\
E & LoRA ($r{=}8$) & Can low-rank adaptation preserve the encoder while adapting? \\

\bottomrule
\end{tabular}
\end{table}
